cmake_minimum_required(VERSION 3.16)
project(ParkingTrajectoryReplanner LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The supplied protected backend targets Windows x64; no Linux binary is bundled.")
endif()
add_executable(parallel_stitching parallel_stitching.cpp)
target_compile_features(parallel_stitching PRIVATE cxx_std_17)
set_target_properties(parallel_stitching PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
get_filename_component(PARKING_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
target_compile_definitions(parallel_stitching PRIVATE PARKING_SOURCE_ROOT="${PARKING_ROOT}")
if(MINGW)
  target_link_options(parallel_stitching PRIVATE -municode -static-libgcc -static-libstdc++)
elseif(MSVC)
  target_compile_options(parallel_stitching PRIVATE /utf-8)
endif()
